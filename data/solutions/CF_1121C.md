## 思路
将求每个树的代价总和转换为：求每条边出现在多少棵树中，再累加贡献。
排序后预处理前缀和 sum 与前后缀乘积 pre/suf，枚举边 i 计算权值与差值乘积并取模累加。

## 代码
```cpp
#include<bits/stdc++.h>
using namespace std;
using ll=long long;
const ll MOD=998244353;
//标签：计数 + 树 + 前缀技巧
/*将求每个树的代价的总和转换为求每条边出现在几个树再累加*/ 
void solve(){
	int n;
	cin>>n;
	vector<ll> a(n);
	vector<ll> sum(n); 
	ll ans=0;
	for(int i=0;i<n;i++) cin>>a[i];
	sort(a.begin(),a.end());
	sum[n-1]=a[n-1];
	for(int i=n-2;i>=0;i--){
		sum[i]=sum[i+1]+a[i];
	}
	vector<ll> pre(n,1),suf(n,1);
	for(int i=0;i<n-1;i++){
		pre[i+1]=pre[i]*(n-i-1)%MOD;
	}
	for(int i=n-2;i>=0;i--){
		suf[i]=suf[i+1]*(n-i-1)%MOD;
	}
	for(int i=0;i<n-1;i++){
		ll diff=((sum[i+1]-(n-i-1)%MOD*a[i])%MOD+MOD)%MOD;
		ll w=pre[i]*suf[i+1]%MOD;
		ll cost=w*diff%MOD;
		ans=(ans+cost)%MOD;
	}
	cout<<ans<<'\n';
}
int main(){
	int t;
	cin>>t;
	while(t--){
		solve();
	}
}
```
